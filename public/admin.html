async function loadAppointments() { try { const response = await fetch('/api/appointments'); const appointments = await response.json(); const tableBody = document.getElementById('appointments-table-body'); tableBody.innerHTML = ''; appointments.forEach(app => { tableBody.innerHTML += ` ${app.name} ${app.phone} ${app.date} (${app.time}) ${app.status} `; }); } catch (error) { console.error('Error loading appointments:', error); } } // Page load hone par call karein loadAppointments();
